.global irq_entry
irq_entry:
    sub     sp, sp, #(16 * 17)
    stp     x0,  x1,  [sp, #16 *  0]
    stp     x2,  x3,  [sp, #16 *  1]
    stp     x4,  x5,  [sp, #16 *  2]
    stp     x6,  x7,  [sp, #16 *  3]
    stp     x8,  x9,  [sp, #16 *  4]
    stp     x10, x11, [sp, #16 *  5]
    stp     x12, x13, [sp, #16 *  6]
    stp     x14, x15, [sp, #16 *  7]
    stp     x16, x17, [sp, #16 *  8]
    stp     x18, x19, [sp, #16 *  9]
    stp     x20, x21, [sp, #16 * 10]
    stp     x22, x23, [sp, #16 * 11]
    stp     x24, x25, [sp, #16 * 12]
    stp     x26, x27, [sp, #16 * 13]
    stp     x28, x29, [sp, #16 * 14]
    str     x30,      [sp, #16 * 15]
    // ★ ELR_EL1 / SPSR_EL1 も退避する。この核は割り込みハンドラの中で
    //    文脈切替(proc_preempt -> proc_resched -> ctxsw)を行うため、
    //    「戻り先が未確定の例外」が同時に複数存在しうる。保存しないと
    //    後から入った例外が上書きし、別の戻り先へ eret して計算が壊れる
    //    （実測: 計算プロセス2本以上のときだけ結果が毎回変わった）。
    mrs     x0,  elr_el1
    mrs     x1,  spsr_el1
    stp     x0,  x1,  [sp, #16 * 16]

    bl      irq_dispatch_c

    ldp     x0,  x1,  [sp, #16 * 16]     // 先に ELR/SPSR を戻す（x0/x1 はこの後で上書きされる）
    msr     elr_el1,  x0
    msr     spsr_el1, x1
    ldp     x0,  x1,  [sp, #16 *  0]
    ldp     x2,  x3,  [sp, #16 *  1]
    ldp     x4,  x5,  [sp, #16 *  2]
    ldp     x6,  x7,  [sp, #16 *  3]
    ldp     x8,  x9,  [sp, #16 *  4]
    ldp     x10, x11, [sp, #16 *  5]
    ldp     x12, x13, [sp, #16 *  6]
    ldp     x14, x15, [sp, #16 *  7]
    ldp     x16, x17, [sp, #16 *  8]
    ldp     x18, x19, [sp, #16 *  9]
    ldp     x20, x21, [sp, #16 * 10]
    ldp     x22, x23, [sp, #16 * 11]
    ldp     x24, x25, [sp, #16 * 12]
    ldp     x26, x27, [sp, #16 * 13]
    ldp     x28, x29, [sp, #16 * 14]
    ldr     x30,      [sp, #16 * 15]
    add     sp, sp, #(16 * 17)
    eret
